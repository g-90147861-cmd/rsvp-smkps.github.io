Panduan Hubung Borang ke Google Sheet (2 Minit)

  1. Buka Google Sheet baharu dan namakan header baris pertama: Tarikh | Nama Penuh | Organisasi | Status | Telefon | Bil Pax | Catatan
  2. Klik menu Extensions > Apps Script.
  3. Padamkan semua kod asal & tampal (*paste*) kod skrip di bawah:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.timestamp || new Date(),
      data.fullName,
      data.organization,
      data.attendanceStatus,
      data.phoneNumber || '-',
      data.paxCount || '-',
      data.notes || '-'
    ]);
    return ContentService.createTextOutput(JSON.stringify({"result":"success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result":"error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
